# A VERSION SPEC, not a git URL, and the difference is not cosmetic.
# A URL pin forces pip to reinstall that exact source even when a
# newer, satisfying jaeger-os is already present — so installing this
# module would DOWNGRADE the framework under a running app and take
# its topics with it.
#
# A version spec leaves a satisfying install alone. If nothing
# satisfies it, pip says so plainly and setup.sh (or the README) says
# where to get it: JaegerOS is not on PyPI yet.
#
#   pip install "jaeger-os>=0.10
#
# REQUIRES the hierarchical topic contract. Below 0.10 the topics this
# module declares do not exist, and the failure is a node that
# subscribes fine and then never receives anything.
jaeger-os>=0.10

# Decoders: PIL for image/gif/sprite, numpy for bitmap/sprite/math + the
# procedural gfx layer. Both are top-level hard imports, so both are
# declared in module.yaml's requires_libraries.
pillow>=10.0
numpy>=1.24

# NOT listed: imageio / imageio-ffmpeg. video_adapter imports them lazily
# inside its methods, so the package imports fine without them and only
# video playback degrades. Install with: pip install "jaeger-animation[video]"
pydantic>=2.0
